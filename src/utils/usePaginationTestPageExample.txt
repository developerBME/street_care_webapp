import React, { useState, useEffect, useRef, useReducer } from "react";

// ==========================================
// 1. HELPERS & THE MOCK PAGINATION SERVICE
// ==========================================
const delay = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

const createDummySnapshots = (pgNo, searchText, filterField) => [
  {
    id: `doc-${pgNo}-1`,
    data: () => ({
      name: `Item ${pgNo}A`,
      matchedSearch: searchText || "none",
      sortedBy: filterField && Object.keys(filterField).length ? JSON.stringify(filterField) : "none",
    }),
  },
  {
    id: `doc-${pgNo}-2`,
    data: () => ({
      name: `Item ${pgNo}B`,
      matchedSearch: searchText || "none",
      sortedBy: filterField && Object.keys(filterField).length ? JSON.stringify(filterField) : "none",
    }),
  },
];

export const mockPaginationService = async ({ baseQuery, pgParams, pgChckpnts }) => {
  const { pgNo, searchText, filterField } = pgParams;
  
  // Safely verify if checkpoints exist and are a plain object
  const updatedCheckpoints = pgChckpnts && typeof pgChckpnts === "object" && !Array.isArray(pgChckpnts) 
    ? { ...pgChckpnts } 
    : {};

  // Build the historical snapshot collection incrementally up to current page
  for (let i = 0; i <= pgNo; i++) {
    if (!updatedCheckpoints[i]) {
      updatedCheckpoints[i] = createDummySnapshots(i, searchText, filterField);
    }
  }

  return {
    pgQuery: async () => {
      await delay(1000); // Simulated 1-second network latency
      return {
        docs: updatedCheckpoints[pgNo],
      };
    },
    pgCheckpoints: updatedCheckpoints,
  };
};

// ==========================================
// 2. STATE MANAGER (REDUCER)
// ==========================================
const reducerFn = (state, action) => {
  switch (action.type) {
    case "SET SEARCHTEXT":
      return { ...state, searchText: action.payload, pgNo: 0 }; // Reset page on query shift
    case "SET SORT FIELD":
      return { ...state, filterField: action.payload, pgNo: 0 }; // Reset page on sort shift
    case "SET PAGE":
      return { ...state, pgNo: action.payload };
    default:
      return state;
  }
};

// ==========================================
// 3. CUSTOM HOOK DEFINITION
// ==========================================
const usePagination = ({ baseQuery, pageCheckpoints = {} }) => {
  const [hookState, setHookState] = useState({
    data: null,
    state: "Loading",//Loading, Success, Failure//
    error: null,
  });

  const pgCheckpointsRef = useRef(pageCheckpoints);
  
  const [pgParams, dispatch] = useReducer(reducerFn, {
    pgNo: 0,
    searchText: "",
    filterField: {},//lastModifiedTS ,asc or desc 
  });

  //filerField to be of format field: value eg:- lastModifiedTimestamp:'asc';

  const getPage = (pgNo) => {
    setHookState({ state: "Loading", data: null, error: null });
    dispatch({ type: "SET PAGE", payload: pgNo });
  };

  const setSearchText = (searchText) => {
    setHookState({ state: "Loading", data: null, error: null });
    dispatch({ type: "SET SEARCHTEXT", payload: searchText });
    pgCheckpointsRef.current = {}; // Flush chckpnts when criteria variables rewrite
  };

  const setFilterField = (filters) => {
    setHookState({ state: "Loading", data: null, error: null });
    dispatch({ type: "SET SORT FIELD", payload: filters });
    pgCheckpointsRef.current = {}; // Flush chckpnts when criteria variables rewrite
  };

  const pgTriggerFns = { getPage, setFilterField, setSearchText };

  useEffect(() => {
    let isMounted = true;
    
    const fetchData = async () => {
      try {
        // Core structural fix: Packed into a single input object configuration
        const { pgQuery, pgCheckpoints } = await mockPaginationService({
          baseQuery,
          pgParams,
          pgChckpnts: pgCheckpointsRef.current,
        });

        if (!isMounted) return;
        pgCheckpointsRef.current = pgCheckpoints;
        
        const result = await pgQuery();
        
        if (isMounted) {
          setHookState({ state: "Success", data: result, error: null });
        }
      } catch (e) {
        if (isMounted) {
          console.error("Error Fetching Docs:", e);
          setHookState({ state: "Error", data: null, error: e });
        }
      }
    };

    fetchData();
    return () => { isMounted = false; }; 
  }, [pgParams, baseQuery]);

  return { hookState, pgTriggerFns, pgParams, currentCheckpoints: pgCheckpointsRef.current };
};
